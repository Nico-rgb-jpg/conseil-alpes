! SW-ETG2 - Switch d'acces, 2e etage (Cisco 2960-24TT)
! Fa0/1-10   : VLAN 30 Technique  (PC-TECH1 sur Fa0/1, PC-TECH2 sur Fa0/2)
! Fa0/11-15  : VLAN 40 Invites    (LAPTOP-INV1 sur Fa0/11)
! Fa0/16-24  : non utilises
! G0/1       : trunk vers SW-CORE G0/2
! G0/2       : non utilise

enable
configure terminal

hostname SW-ETG2
no ip domain-lookup
service password-encryption
enable secret Alpes@2026
banner motd # Acces reserve au personnel autorise d'Alpes Conseil #
username admin privilege 15 secret Admin@2026
ip domain-name alpes.local

vtp mode transparent
vlan 30
 name TECHNIQUE
vlan 40
 name INVITES
vlan 99
 name GESTION
vlan 999
 name INUTILISES
 exit

! --- Uplink ---
interface GigabitEthernet0/1
 description Trunk vers SW-CORE G0/2
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,40,99,999
 switchport nonegotiate
 no shutdown
 exit

! --- Ports utilisateurs ---
interface range FastEthernet0/1 - 10
 description Postes Technique
 switchport mode access
 switchport access vlan 30
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit

interface range FastEthernet0/11 - 15
 description Invites
 switchport mode access
 switchport access vlan 40
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit

interface range FastEthernet0/16 - 24
 description Non utilise
 switchport mode access
 switchport access vlan 999

 exit

 switchport mode access
 switchport access vlan 999

 exit
